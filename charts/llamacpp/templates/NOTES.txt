1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.tls.host | default .Values.ingress.host | list }}
  http{{ if $.Values.ingress.tls.enabled }}s{{ end }}://{{ $host }}
{{- end }}
{{- else if eq .Values.service.type "ClusterIP" }}
  kubectl port-forward svc/{{ include "llamacpp.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }} --namespace {{ .Release.Namespace }}

  Then open your browser or API client at:
  http://127.0.0.1:{{ .Values.service.port }}
{{- else if eq .Values.service.type "NodePort" }}
  export NODE_PORT=$(kubectl get svc {{ include "llamacpp.fullname" . }} --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}")
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "llama-server URL: http://$NODE_IP:$NODE_PORT"
{{- else if eq .Values.service.type "LoadBalancer" }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  export SERVICE_IP=$(kubectl get svc {{ include "llamacpp.fullname" . }} --namespace {{ .Release.Namespace }} -o jsonpath="{.status.loadBalancer.ingress[0].ip}")
  echo "llama-server URL: http://$SERVICE_IP:{{ .Values.service.port }}"
{{- end }}

2. Verify the server is healthy:
  curl http://127.0.0.1:{{ .Values.service.port }}/health

{{- if .Values.llamacpp.model }}
3. Before running inference, ensure a model file is placed at:
  {{ .Values.llamacpp.model }}

   You can copy a model into the PVC using:
  kubectl cp /path/to/model.gguf {{ .Release.Namespace }}/$(kubectl get pod -l app.kubernetes.io/instance={{ .Release.Name }} -n {{ .Release.Namespace }} -o jsonpath="{.items[0].metadata.name}"):/models/model.gguf
{{- end }}
